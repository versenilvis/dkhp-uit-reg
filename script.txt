// ==UserScript==
// @name         Auto ĐKHP UIT (Reload Mode)
// @namespace    http://tampermonkey.net/
// @version      Final
// @description  Tự động chọn lớp, nếu lỗi hoặc không thấy lớp -> Tự Reload
// @author       Gemini Modified
// @match        https://dkhp.uit.edu.vn/app/reg
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // ================= KHAI BÁO MÔN =================
    var monDangKy = `
    CS221.Q21.1
    CS221.Q21
    CE103.Q25
    CE103.Q25.2
    `;
    // ================================================

    var delay = 1000;       // Thời gian chờ giữa các thao tác click
    var reloadDelay = 3000; // Thời gian chờ trước khi reload (để bạn kịp nhìn lỗi)

    // --- LOGGING ---
    var successLog = (msg) => console.log('%c' + msg, 'font-weight:bold; color:green; font-size:14px;');
    var errorLog = (msg) => console.log('%c' + msg, 'font-weight:bold; color:red; font-size:14px;');
    var infoLog = (msg) => console.log('%c' + msg, 'font-weight:bold; color:blue; font-size:14px;');
    var warningLog = (msg) => console.log('%c' + msg, 'font-weight:bold; color:orange; font-size:14px;');

    var isRunning = true;
    var daGuiYeuCau = false;

    // --- KIỂM TRA TRẠNG THÁI TỰ ĐỘNG CHẠY SAU KHI RELOAD ---
    if (sessionStorage.getItem('AUTO_DKHP') === 'true') {
        infoLog('🔄 Đã phát hiện Reload - Script sẽ tự chạy lại sau 2s...');
        setTimeout(guiLienTuc, 2000);
    } else {
        // Nếu lần đầu bật script, hãy chạy hàm startManually() hoặc gõ guiLienTuc()
        infoLog('💡 Script đã sẵn sàng. Gõ guiLienTuc() để bắt đầu hoặc script sẽ tự chạy nếu bạn đã set lệnh.');
        // Bỏ comment dòng dưới nếu muốn chạy ngay lập tức khi vào trang:
        guiLienTuc();
    }

    // --- HÀM XỬ LÝ RELOAD ---
    function triggerReload() {
        if (!isRunning) return;
        warningLog(`⏳ Sẽ reload trang sau ${reloadDelay/1000} giây để cập nhật slot...`);

        // Đặt cờ để lần sau vào lại biết là đang chạy dở
        sessionStorage.setItem('AUTO_DKHP', 'true');

        setTimeout(() => {
            location.reload();
        }, reloadDelay);
    }

    function timHangMon() {
        var listMon = monDangKy.trim().split('\n').map(it => it.trim()).filter(it => it);
        var allRows = [...document.querySelectorAll('tbody tr')];
        return allRows.filter(row => {
            var maMon = row.querySelector('td:nth-child(2)')?.textContent?.trim();
            return listMon.includes(maMon);
        });
    }

    function chonVaDangKy() {
        try {
            // Reset check cũ
            document.querySelectorAll('input[type="checkbox"]:checked').forEach(cb => cb.click());

            // Tìm lớp
            var rowsCanDangKy = timHangMon();

            // QUAN TRỌNG: Nếu không thấy lớp -> Có thể do chưa F5 -> RELOAD
            if (rowsCanDangKy.length === 0) {
                errorLog('⚠ Không tìm thấy lớp nào trong danh sách hiển thị.');
                triggerReload();
                return false;
            }

            var daChonDuoc = false;
            // Chọn lớp
            rowsCanDangKy.forEach((row) => {
                var checkbox = row.querySelector('input[type="checkbox"]');
                // Chỉ chọn nếu checkbox tồn tại và chưa bị disable (hết slot)
                if (checkbox && !checkbox.disabled && !checkbox.checked) {
                    checkbox.click();
                    var tenLop = row.querySelector('td:nth-child(2)')?.textContent?.trim();
                    successLog(`✓ Đã chọn: ${tenLop}`);
                    daChonDuoc = true;
                }
            });

            if (!daChonDuoc) {
                errorLog('⚠ Tìm thấy môn nhưng checkbox bị ẩn/disabled (Hết slot).');
                triggerReload();
                return false;
            }

            // Tìm nút Đăng ký (Cập nhật selector mạnh hơn)
            var btnDangKy = [...document.querySelectorAll('button, input[type="submit"]')].find(btn =>
                (btn.innerText || btn.value).toLowerCase().includes('đăng ký')
            );

            if (btnDangKy) {
                btnDangKy.click();
                infoLog('→ Đã gửi yêu cầu đăng ký...');
                daGuiYeuCau = true; // Chặn gửi lặp lại để chờ fetch/alert
                return true;
            } else {
                errorLog('❌ Không tìm thấy nút Đăng ký!');
                triggerReload();
                return false;
            }

        } catch (error) {
            errorLog('❌ Lỗi code: ' + error.message);
            triggerReload();
            return false;
        }
    }

    function guiLienTuc() {
        if (daGuiYeuCau) return;

        // Đánh dấu là đang chạy để Auto-Reload hoạt động
        sessionStorage.setItem('AUTO_DKHP', 'true');

        var ketQua = chonVaDangKy();

        // Nếu chưa click được nút đăng ký, thử lại sau delay (trường hợp web lag hiển thị chậm)
        if (!ketQua) {
            setTimeout(guiLienTuc, delay);
        }
    }

    // --- OVERRIDE FETCH (Xử lý kết quả từ Server) ---
    var originalFetch = window.fetch;
    window.fetch = function(...args) {
        return originalFetch.apply(this, args).then(response => {
            response.clone().json().then(data => {
                // THÀNH CÔNG
                if (data.success || (data.message && data.message.toLowerCase().includes('thành công'))) {
                    successLog('🎉🎉🎉 ĐĂNG KÝ THÀNH CÔNG! 🎉🎉🎉');
                    successLog('📋 Data: ' + JSON.stringify(data));

                    // Xóa cờ Auto Run để script dừng hẳn
                    isRunning = false;
                    daGuiYeuCau = false;
                    sessionStorage.removeItem('AUTO_DKHP');
                    alert('ĐĂNG KÝ THÀNH CÔNG!');

                // THẤT BẠI / LỖI -> RELOAD
                } else {
                    errorLog('❌ KẾT QUẢ THẤT BẠI: ' + (data.message || 'Lỗi không xác định'));
                    triggerReload();
                }
            }).catch(() => {});
            return response;
        });
    };

    // --- OVERRIDE ALERT (Xử lý Alert truyền thống) ---
    var originalAlert = window.alert;
    window.alert = function(msg) {
        console.log('%c⚠ Alert: ' + msg, 'font-size:14px; color:orange;');
        var m = msg.toLowerCase();

        if (m.includes('thành công')) {
            successLog('🎉 THÀNH CÔNG (Alert)!');
            isRunning = false;
            sessionStorage.removeItem('AUTO_DKHP');
        } else if (m.includes('thất bại') || m.includes('lỗi') || m.includes('không')) {
            errorLog('❌ Lỗi Alert: ' + msg);
            triggerReload();
        }
        return true; // Chặn popup
    };

    // Hàm dừng khẩn cấp
    window.stopDKHP = function() {
        isRunning = false;
        daGuiYeuCau = true;
        sessionStorage.removeItem('AUTO_DKHP');
        successLog('⏹ Đã dừng script hoàn toàn.');
    };

})();
